  curl -s -X POST http://localhost:{{ .Values.service.port }}/narrate \

    {{ end }}-H 'Content-Type: application/json' \
    -d '{"text":"Once upon a time there was a rabbit.","voice":"preschool"}' \
    --output story.wav
